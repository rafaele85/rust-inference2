// === src/activation.rs ===
const GELU_COEFF: f32 = 0.044715;
const HALF: f32 = 0.5;

const SQRT_2_OVER_PI: f32 = 0.7978845608;
pub fn get_gelu_scalar(x: f32) -> f32 {
    let inner = SQRT_2_OVER_PI * (x + GELU_COEFF * x.powi(3));
    let result = HALF * x * (1.0 + inner.tanh());

    result
}

pub fn gelu(input: &[f32]) -> Vec<f32> {
    input.iter()
        .map(|&x| get_gelu_scalar(x))
        .collect()
}

pub fn softmax(input: &[f32]) -> Vec<f32> {
    let max_val = input.iter().cloned().fold(f32::NEG_INFINITY, f32::max);
    let exps: Vec<f32> = input.iter().map(|&x| (x - max_val).exp()).collect();
    let sum: f32 = exps.iter().sum();
    let result = exps.iter().map(|&x| x / sum).collect();

    result
}

pub fn argmax(values: &[f32]) -> usize {
    let mut max_idx = 0;
    let mut max_val: f32 = f32::NEG_INFINITY;

    for (i, &v) in values.iter().enumerate() {
        if v > max_val {
            max_val = v;
            max_idx = i;
        }
    }

    max_idx
}

pub struct QKV {
    pub q: Vecf32,
    pub k: Vecf32,
    pub v: Vecf32,
}
pub fn extract_qkv(qkv: &Arrf32) -> QKV {
    let dim = qkv.len() / 3;
    let q = qkv[0..dim].to_vec();
    let k = qkv[dim..dim * 2].to_vec();
    let v = qkv[dim * 2..dim * 3].to_vec();

    QKV {q, k, v}
}

pub struct ComputeScoreArgs<'a> {
    all_q: &'a Arrf32,
    q_offset: usize,
    all_k: &'a Arrf32,
    k_offset: usize,
    head_dim: usize,
}
pub fn compute_score(args: ComputeScoreArgs) -> f32 {
    let ComputeScoreArgs {all_q, all_k, q_offset, k_offset, head_dim} = args;
    let mut dot = 0.0f32;
    for i in 0..head_dim {
        dot += all_q[q_offset + i] * all_k[k_offset + i];
    }

    let score = dot / (head_dim as f32).sqrt();

    score
}

pub struct CachedComputeScores<'a> {
    pub cache_len: usize,
    pub k_cache: &'a PositionKVVectors,
    pub head_dim: usize,
    pub h_offset: usize,
    pub q: &'a Arrf32,
}
fn cached_compute_scores(args: CachedComputeScores) -> Vecf32 {
    let CachedComputeScores {cache_len, k_cache, head_dim, h_offset, q} = args;

    let mut scores = vec![0.0f32; cache_len];
    for prev in 0..cache_len {
        let all_k = &k_cache[prev];
        let score = compute_score(ComputeScoreArgs {
            head_dim, all_q: q, all_k, q_offset: h_offset, k_offset: h_offset
        });
        scores[prev] = score;
    }

    scores
}


pub struct CachedAttentionArgs<'a> {
    pub q: &'a Arrf32,
    pub k_cache: &'a PositionKVVectors,
    pub v_cache: &'a PositionKVVectors,
    pub cache_len: usize,
    pub n_heads: usize,
    pub dim: usize,
    pub head_dim: usize,
}
pub fn cached_attention(args: CachedAttentionArgs) -> Vecf32 {
    let CachedAttentionArgs {q, k_cache, v_cache, n_heads, dim, head_dim, cache_len} = args;

    let mut output = vec![0.0f32; dim];

    for h in 0..n_heads {
        let h_offset = h * head_dim;
        let scores = cached_compute_scores(CachedComputeScores {
            cache_len, k_cache, head_dim, h_offset, q
        });

        let probs = softmax(&scores);

        scale_and_add_multi(ScaleAndAddMultiArgs {
            output: &mut output, sources: v_cache, num_iterations: head_dim, start_offset: h_offset, scales: &probs, seq_len: cache_len
        });
    }
    output
}

pub fn combine_embedding_weights(wte_row: &[f32], wpe_row: &[f32]) -> Vec<f32> {
    let result = wte_row.iter()

use std::sync::Arc;
use cudarc::cublas::CudaBlas;
use cudarc::cublas::result::sgemm;
use cudarc::cublas::sys::cublasOperation_t::{CUBLAS_OP_N, CUBLAS_OP_T};
use cudarc::driver::{CudaDevice, CudaSlice, DevicePtr, DevicePtrMut};
use crate::common_types::{Arrf32, Vecf32};

pub type CudaArrf32 = CudaSlice<f32>;

pub struct Gpu {
    pub dev: Arc<CudaDevice>,
    pub blas: CudaBlas,
}

impl Gpu {
    pub fn new(gpu_index: usize) -> Self {
        let dev = CudaDevice::new(gpu_index).unwrap();
        let blas = CudaBlas::new(dev.clone()).unwrap();
        Gpu { dev, blas }
    }

    fn sgemm(&self, input: &Arrf32, d_weight: &CudaArrf32, in_dim: usize, out_dim: usize, transpose: bool) -> Vecf32 {
        let d_input = self.dev.htod_sync_copy(input).unwrap();
        let mut d_output = self.dev.alloc_zeros::<f32>(out_dim).unwrap();

        let handle = *self.blas.handle();
        let trans = if transpose { CUBLAS_OP_T } else { CUBLAS_OP_N };
        let transb = CUBLAS_OP_N;
        let m = out_dim as i32;
        let n = 1;
        let k = in_dim as i32;
        let alpha = &1.0f32 as *const f32;
        let a = *d_weight.device_ptr() as *const f32;
        let lda = if transpose { in_dim as i32 } else { out_dim as i32 };
        let b = *d_input.device_ptr() as *const f32;
        let ldb = in_dim as i32;
        let beta = &0.0f32 as *const f32;
        let c = *d_output.device_ptr_mut() as *mut f32;
        let ldc = out_dim as i32;
        unsafe {
            sgemm(
                handle,
                trans, transb,
                m, n, k,
                alpha,
                a, lda, b, ldb,
                beta,
                c, ldc
            ).unwrap();
        }
        let result = self.dev.dtoh_sync_copy(&d_output).unwrap();
        result
    }

    pub fn matmul(&self, input: &Arrf32, weight: &CudaArrf32, in_dim: usize, out_dim: usize) -> Vec<f32> {
        self.sgemm(input, weight, in_dim, out_dim, false)
    }

    pub fn matmul_transposed(&self, input: &Arrf32, weight: &CudaArrf32, in_dim: usize, out_dim: usize) -> Vec<f32> {
        self.sgemm(input, weight, in_dim, out_dim, true)
    }

    pub fn upload(&self, data: &Arrf32) -> CudaSlice<f32> {
        self.dev.htod_sync_copy(data).unwrap()
    }
}

use crate::common_types::{Arrf32, Vecf32, Weight};
use crate::gpu::Gpu;
use crate::matmul::matmul;
use crate::norm::layer_norm;

pub fn linear_projection(input: &[f32], weight: &Weight, bias: &[f32], out_dim: usize, gpu: &Gpu) -> Vec<f32> {
    let mut output = matmul(input, weight, out_dim, gpu);
    for i in 0..out_dim {
        output[i] += bias[i];
    }

    output
}

pub fn add_to_vector(input: &mut [f32], vector_to_add: &[f32]) -> () {
    for i in 0..input.len() {
        input[i] += vector_to_add[i];
    }
}

pub struct ProjectAndResidualArgs<'a> {
    pub activation_weights: &'a [f32],
    pub projection_weights: &'a Weight,
    pub projection_bias: &'a [f32],
    pub dim: usize,
    pub hidden: &'a mut [f32],
    pub gpu: &'a Gpu,
}

pub fn project_and_residual(args: ProjectAndResidualArgs) -> () {
    let projected = linear_projection(&args.activation_weights, &args.projection_weights, &args.projection_bias, args.dim, args.gpu);
    add_to_vector(args.hidden, &projected);
}

pub struct NormalizeAndProject<'a> {
    pub hidden: &'a Arrf32,
    pub norm_weights: &'a Vecf32,
    pub norm_bias: &'a Vecf32,
    pub projection_weights: &'a Weight,
    pub projection_bias: &'a Vecf32,
    pub projection_dim: usize,
    pub gpu: &'a Gpu
}

pub fn normalize_and_project(args: NormalizeAndProject) -> Vecf32 {
    let mut normalized = layer_norm(args.hidden, args.norm_weights, args.norm_bias);
    let result = linear_projection(&mut normalized, args.projection_weights, args.projection_bias, args.projection_dim, args.gpu);

    result
}

pub struct ScaleAndAddArgs<'a> {
    pub output: &'a mut Vecf32,
    pub source: &'a Vecf32,
    pub scale: f32,
    pub start_offset: usize,
    pub num_iterations: usize,
}
pub fn scale_and_add(args: ScaleAndAddArgs) {
    let ScaleAndAddArgs {output, source, scale, start_offset, num_iterations} = args;
    for i in 0..num_iterations {
        output[start_offset + i] += scale * source[start_offset + i];
    }
}

pub struct ScaleAndAddMultiArgs<'a> {
    pub output: &'a mut Vecf32,
    pub sources: &'a Vec<Vecf32>,
    pub scales: &'a Vecf32,
    pub start_offset: usize,
    pub num_iterations: usize,
    pub seq_len: usize,
}
pub fn scale_and_add_multi(args: ScaleAndAddMultiArgs) {
    let ScaleAndAddMultiArgs {seq_len, scales, sources, output, num_iterations, start_offset} = args;
    for i in 0..seq_len {
        let scale = scales[i];
        let source = &sources[i];
        scale_and_add(ScaleAndAddArgs {output, source, num_iterations, start_offset, scale});
    }

}

pub struct ComputeLogitsArgs<'a> {
    pub hidden: &'a [f32],
    pub ln_f_w: &'a [f32],
    pub ln_f_b: &'a [f32],
    pub wte: &'a Weight,
    pub last_pos: usize,
    pub dim: usize,
    pub vocab_size: usize,
    pub gpu : &'a Gpu,
}

pub fn compute_logits(args: ComputeLogitsArgs) -> usize {
    let ComputeLogitsArgs {hidden, ln_f_w, ln_f_b, wte, last_pos, dim, vocab_size, gpu} = args;
    let start = last_pos * dim;
    let end = start + dim;
    let last_h = &hidden[start..end];
    let ln_out = layer_norm(last_h, ln_f_w, ln_f_b);
    let logits = matmul_transposed(&ln_out, wte, vocab_size, gpu);
    let top_id = argmax(&logits);

    top_id
}

// === src/norm.rs ===
pub fn layer_norm(input: &[f32], weight: &[f32], bias: &[f32] ) -> Vec<f32> {
    let n = input.len() as f32;
    let mean: f32 = input.iter().sum::<f32>() / n;
    let variance: f32 = input.iter().map(|x| (x - mean).powi(2)).sum::<f32>() / n;
    let eps = 1e-5f32;
    let std = (variance + eps).sqrt();

    let result = input.iter().enumerate().map(|(i, &x)| {
        weight[i] * (x - mean) / std + bias[i]
    }).collect();

    result
}

use std::collections::HashMap;
use std::fs::File;
use std::io::{Read, Seek};
use std::io::SeekFrom::Start;
use crate::common_types::Weight;
use crate::gpu::Gpu;

struct TensorEntry {
    offsets: [u64; 2],
    shape: Vec<usize>
}
pub struct SafeTensor {
    file: File,
    data_offset: u64,
    entries: HashMap<String, TensorEntry>
}

impl SafeTensor {
    pub fn open(path: &str) -> Self {
        let mut file = File::open(path).expect("Failed to open file");
        let mut buf = [0u8; 8];
        file.read_exact(&mut buf).expect("Failed to read file");
        let header_len = u64::from_le_bytes(buf);
        let mut header_bytes = vec![0u8; header_len as usize];
        file.read_exact(&mut header_bytes).expect("Failed to read file");
        let header_str = String::from_utf8(header_bytes).expect("Invalid UTF-8");
        let json: HashMap<String, serde_json::Value> = serde_json::from_str(&header_str).expect("Invalid JSON");

        let mut entries = HashMap::new();
        for (name, value) in &json {
            if name == "__metadata__" {
                continue;
            }
            let offsets: Vec<u64> = serde_json::from_value(value["data_offsets"].clone()).unwrap();
            let shape: Vec<usize> = serde_json::from_value(value["shape"].clone()).unwrap();
            let tensor_entry = TensorEntry {
                offsets: [offsets[0], offsets[1]],
                shape
            };
            entries.insert(name.clone(), tensor_entry);
        }

        let data_offset = 8 + header_len;
        SafeTensor { file, data_offset, entries }
    }

    pub fn load_tensor(&mut self, name: &str) -> Vec<f32> {
        if !self.entries.contains_key(name) {
            println!("NOT FOUND KEY {}", name);
        }
        let entry = &self.entries[name];
        let start = self.data_offset + entry.offsets[0];
        let byte_len = (entry.offsets[1] - entry.offsets[0]) as usize;

        let mut bytes = vec![0u8; byte_len];
        self.file.seek(Start(start)).unwrap();
        self.file.read_exact(&mut bytes).expect("Failed to read file");

        let result = bytes.chunks_exact(4).map(|c| f32::from_le_bytes(c.try_into().unwrap())).collect();

        result
    }
}


fn load_weight_bias(model: &mut SafeTensor, name: &str, layer: usize) -> (Vec<f32>, Vec<f32>) {
    let prefix = format!("h.{}", layer);
    let w = model.load_tensor( &format!("{}.{}.weight", prefix, name));
    let b = model.load_tensor(&format!("{}.{}.bias", prefix, name));

    (w, b)
}

pub struct Layer {
    pub ln1_w: Vec<f32>,
    pub ln1_b: Vec<f32>,
    pub proj_w: Weight,
    pub proj_b: Vec<f32>,
    pub ln2_w: Vec<f32>,
    pub ln2_b: Vec<f32>,
    pub ffn_up_w: Weight,
    pub ffn_up_b: Vec<f32>,
    pub ffn_down_w: Weight,
    pub ffn_down_b: Vec<f32>,
    pub qkv_w: Weight,
    pub qkv_b: Vec<f32>,
}

pub fn load_layer(model: &mut SafeTensor, layer: usize, gpu: &Gpu) -> Layer {
    let (ln1_w, ln1_b) = load_weight_bias(model, "ln_1", layer);
    let (qkv_w, qkv_b) = load_weight_bias(model, "attn.c_attn", layer);
    let (proj_w, proj_b) = load_weight_bias(model, "attn.c_proj", layer);
    let (ln2_w, ln2_b) = load_weight_bias(model, "ln_2", layer);
    let (ffn_up_w, ffn_up_b) = load_weight_bias(model, "mlp.c_fc", layer);
    let (ffn_down_w, ffn_down_b) = load_weight_bias(model, "mlp.c_proj", layer);

    Layer { ln1_w, ln1_b, qkv_w: gpu.upload(&qkv_w), qkv_b, proj_w: gpu.upload(&proj_w), proj_b, ln2_w, ln2_b, ffn_up_w: gpu.upload(&ffn_up_w), ffn_up_b, ffn_down_w: gpu.upload(&ffn_down_w), ffn_down_b }
}

pub fn load_all_layers(model: &mut SafeTensor, nlayers: usize, gpu: &Gpu) -> Vec<Layer> {

    let mut layers = Vec::with_capacity(nlayers);
    for i in 0..nlayers {
        let layer = load_layer(model, i, gpu);
        layers.push(layer);
    }

    layers
}

pub struct GenerateTokenSingleLayerArgs<'a> {
    pub layer_no: usize,
    pub qkv_size: usize,
    pub n_heads: usize,
    pub dim: usize,
    pub hidden: &'a mut Vecf32,
    pub layer: &'a Layer,
    pub kv_cache: &'a mut KVCache,
    pub head_dim: usize,
    pub ffn_dim: usize,
    pub gpu: &'a Gpu
}

pub fn generate_token_single_layer(args: GenerateTokenSingleLayerArgs) {
    let GenerateTokenSingleLayerArgs {
        hidden, layer, layer_no, qkv_size,
        kv_cache, n_heads, dim, head_dim, ffn_dim, gpu
    } = args;

    let qkv = normalize_and_project(NormalizeAndProject {
        hidden, norm_weights: &layer.ln1_w, norm_bias: &layer.ln1_b,
        projection_weights: &layer.qkv_w, projection_bias: &layer.qkv_b, projection_dim: qkv_size, gpu
    });

    let QKV {q, k, v} = extract_qkv(&qkv);

    kv_cache.append(layer_no, k, v);

    let k_cache = &kv_cache.k[layer_no];
    let v_cache = &kv_cache.v[layer_no];
    let cache_len = k_cache.len();
    let attn_raw = cached_attention( CachedAttentionArgs {
        q: &q,
        k_cache,
        v_cache,
        cache_len,
        n_heads,
        dim,
        head_dim,
    });

    project_and_residual(ProjectAndResidualArgs {
        activation_weights: &attn_raw, projection_weights: &layer.proj_w, projection_bias: &layer.proj_b, dim, hidden, gpu
    });

    let ffn_up = normalize_and_project(NormalizeAndProject {
       hidden, norm_weights: &layer.ln2_w, norm_bias: &layer.ln2_b, projection_weights: &layer.ffn_up_w, projection_bias: &layer.ffn_up_b, projection_dim: ffn_dim, gpu
    });

    let ffn_act = gelu(&ffn_up);

    project_and_residual(ProjectAndResidualArgs {
        activation_weights: &ffn_act, projection_weights: &layer.ffn_down_w, projection_bias: &layer.ffn_down_b, dim, hidden, gpu
    });
}

pub fn generate_token(args: GenerateTokenArgs) -> usize {
    let GenerateTokenArgs {
        token_id, dim, position, wte, wpe, num_layers, layers,
        n_heads, ffn_dim, kv_cache, head_dim, ln_f_w, ln_f_b, wte_gpu,
        vocab_size, gpu
    } = args;

    let wte_start = token_id * dim;
    let wpe_start = position * dim;
    let wte_row = &wte[wte_start..wte_start + dim];
    let wpe_row = &wpe[wpe_start..wpe_start + dim];
    let qkv_size = dim * 3;

    let mut hidden = combine_embedding_weights(wte_row, wpe_row);

    for layer_no in 0..num_layers {
        let layer = &layers[layer_no];
        generate_token_single_layer(GenerateTokenSingleLayerArgs {
            layer_no, layer, qkv_size, hidden: &mut hidden, n_heads, ffn_dim, kv_cache, dim, head_dim, gpu
        });
    }
    let top_id = compute_logits(ComputeLogitsArgs {hidden: &hidden, wte: wte_gpu, ln_f_w, ln_f_b, last_pos: 0, dim, vocab_size, gpu});

    top_id
}

use std::collections::HashMap;
use std::fs;

pub struct Vocab {
    token_to_id: HashMap<String, usize>,
    id_to_token: HashMap<usize, String>,
}

impl Vocab {
    pub fn load(path: &str) -> Self {

        let content = fs::read_to_string(path).expect("Failed to read file");
        let token_to_id: HashMap<String, usize> = serde_json::from_str(&content).expect("Failed to parse file");

        let mut id_to_token = HashMap::new();
        for (token, id) in &token_to_id {
            id_to_token.insert(*id, token.clone());
        }

        Vocab { id_to_token, token_to_id }
    }

    pub fn decode(&self, token_id: usize) -> String {
        let token = &self.id_to_token[&token_id];
        token.replace("Ġ", " ")
    }

    pub fn encode(&self, str: &str) -> Vec<usize> {
        let mut ids = Vec::new();

        let mut remaining = str;

        while !remaining.is_empty() {
            let mut found = false;
            let max_len = remaining.len();
            let range = (1..=max_len).rev();
            for len in range {
                let candidate = remaining[..len].replace(" ", "Ġ");
                let lookup = self.token_to_id.get(&candidate);
                if let Some(id) = lookup {
                    ids.push(*id);
                    remaining = &remaining[len..];
                    found = true;
                    break;
                }
            }
            if !found {
                panic!("Can not tokenize {}", str);
            }
        }

        ids
    }

    pub fn size(&self) -> usize {
        self.id_to_token.len()
    }
}
